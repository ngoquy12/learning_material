Cú Pháp Class Engine Kiểm Tra Đạt Chuẩn CLO/PLO

CourseEvaluationEngine.js 
class CourseEvaluationEngine {
  constructor(courseCode, totalSessions) {
    this.courseCode = courseCode;
    this.totalSessions = totalSessions;
    this.weights = { attendance: 0.20, midtermExam: 0.30, finalPracticalExam: 0.50 };
  }

  calculateFinalScore(attendanceScore, midtermScore, finalPracticalScore) {
    const totalScore = (attendanceScore * this.weights.attendance) +
      (midtermScore * this.weights.midtermExam) +
      (finalPracticalScore * this.weights.finalPracticalExam);
    return Number(totalScore.toFixed(2));
  }

  verifyCLOCompletion(finalScore, attendedSessions) {
    const minSessionsRequired = this.totalSessions * 0.8;
    const isAttendanceValid = attendedSessions >= minSessionsRequired;
    const isPassed = finalScore >= 5.0 && isAttendanceValid;
    return {
      courseCode: this.courseCode, finalScore, attendedSessions, isPassed,
      statusMessage: isPassed ? "ĐẠT CHUẨN ĐẦU RA (CLO/PLO)" : "KHÔNG ĐẠT CHUẨN ĐẦU RA"
    };
  }
}
Input: Điểm (9.0, 7.5, 8.0) | Tham gia: 9/10 buổi
Output: Điểm tổng kết: 8.05 | Trạng thái: ĐẠT CHUẨN ĐẦU RA

BAD PRACTICE GOOD PRACTICE

Sai Lầm Phổ Biến (Bad)

let scoreB = 4;

  // BAD: Bỏ qua kiểm tra điểm danh 80%

PROMPT THIẾU THÔNG TIN PROMPT CHUẨN RCTC

Sai Lầm (Bad Prompt) 

Giải thích cho tôi về phương pháp tự học với cách làm việc nhóm.

[Constraint]: Mỗi kỹ thuật giải thích trong 2 câu, dạng bảng so sánh.

Cú Pháp Xử Lý Layout Metrics & Responsive

LayoutMetrics.js 
const DEVICE_BREAKPOINTS = { MOBILE: 768, TABLET: 1024 };

function calculateLayoutMetrics(viewportWidth) {
  const isMobile = viewportWidth < DEVICE_BREAKPOINTS.MOBILE;
  const isTablet = viewportWidth >= DEVICE_BREAKPOINTS.MOBILE && viewportWidth < DEVICE_BREAKPOINTS.TABLET;

  const maxProductsPerPage = isMobile ? 4 : (isTablet ? 8 : 12);
  const layoutColumns = isMobile ? 1 : (isTablet ? 2 : 4);

  return {
    deviceCategory: isMobile ? "MOBILE" : (isTablet ? "TABLET" : "DESKTOP"),
    maxProductsPerPage: maxProductsPerPage,
    layoutColumns: layoutColumns
  };
}

const mobileTestMetrics = calculateLayoutMetrics(375);
console.log("Mobile (375px):", mobileTestMetrics);
