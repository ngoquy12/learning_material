class CourseEvaluationEngine {
  constructor(courseCode, totalSessions) {
    this.courseCode = courseCode;
    this.totalSessions = totalSessions;
    this.weights = { attendance: 0.20, midtermExam: 0.30, finalPracticalExam: 0.50 };
  }

  calculateFinalScore(attendanceScore, midtermScore, finalPracticalScore) {
    const totalScore = (attendanceScore * this.weights.attendance) +
      (midtermScore * this.weights.midtermExam) +
      (finalPracticalScore * this.weights.finalPracticalExam);
    return Number(totalScore.toFixed(2));
  }

  verifyCLOCompletion(finalScore, attendedSessions) {
    const minSessionsRequired = this.totalSessions * 0.8;
    const isAttendanceValid = attendedSessions >= minSessionsRequired;
    const isPassed = finalScore >= 5.0 && isAttendanceValid;
    return {
      courseCode: this.courseCode, finalScore, attendedSessions, isPassed,
      statusMessage: isPassed ? "ĐẠT CHUẨN ĐẦU RA (CLO/PLO)" : "KHÔNG ĐẠT CHUẨN ĐẦU RA"
    };
  }
}
