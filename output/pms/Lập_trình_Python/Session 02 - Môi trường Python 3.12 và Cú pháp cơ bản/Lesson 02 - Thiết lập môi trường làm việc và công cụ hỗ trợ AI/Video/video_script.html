# Prompt cho AI: Viết hàm tính tổng các số chẵn trong danh sách
def sum_even_numbers(numbers: list[int]) -> int:
    """Tính tổng các số chẵn trong danh sách cho trước."""
    return sum(n for n in numbers if n % 2 == 0)
