# Khai báo danh sách học viên
students = ["An", "Bình", "Châu", "Dũng"]

# Truy xuất phần tử đầu tiên (Index 0)
first_student = students[0]   # Kết quả: "An"

# Truy xuất phần tử cuối cùng (Index -1)
